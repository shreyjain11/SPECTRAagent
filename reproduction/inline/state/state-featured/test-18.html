STATE — WikiPathways cross-ontology check¶

WikiPathways provides another collection of biological pathways linking genes by function. The audit repeated the support comparison using these pathway memberships to check whether the association depended on the choice of annotation source.

Run the six sections in order. Every custom calculation is written in these cells; no external analysis scripts or saved derived tables are required. The analysis uses official STATE released evaluation scores for the HVG and SE representations. The audit does not fine-tune STATE or run fresh inference.

Use Python 3.11 with numpy==1.26.4 pandas==2.2.2 scipy==1.13.1 matplotlib==3.9.4 pyarrow==16.1.0 rdkit==2025.3.6 ipython ipykernel. An existing raw-data cache can be selected with STATE_PRIMARY_INPUTS; each input is checked against its declared SHA256 before reuse.

1. Data source/acquisition¶

The source table specifies pinned score downloads, split definitions, and annotation inputs. Annotation files are distributed in a versioned public data package; their upstream providers are listed alongside the package URL. Only raw annotations are read from that package.

def sha256_file(path):
    h=hashlib.sha256()
    with Path(path).open('rb') as f:
        for block in iter(lambda:f.read(8<<20),b''):h.update(block)
    return h.hexdigest()

def verified(path,entry):
    return path.is_file() and path.stat().st_size==entry['bytes'] and sha256_file(path)==entry['sha256']

def download(entry):
    path=INPUTS/entry['path']
    if verified(path,entry):return path
    path.parent.mkdir(parents=True,exist_ok=True)
    temporary=path.with_name(path.name+'.part')
    for attempt in range(4):
        try:
            request=urllib.request.Request(entry['url'],headers={'User-Agent':'SPECTRA-notebook/1.0'})
            with urllib.request.urlopen(request,timeout=120) as response, temporary.open('wb') as f:
                shutil.copyfileobj(response,f,8<<20)
            if not verified(temporary,entry):raise ValueError('Download checksum mismatch: '+entry['path'])

            return path
        except Exception:

def i_parse_perturbation(value: str):
    """Parse STATE perturbation strings into name/dose/unit when possible."""
    if not isinstance(value, str):
        return (None, np.nan, None)
    s = value.strip()
    try:
        obj = ast.literal_eval(s)
        if isinstance(obj, list) and obj and isinstance(obj[0], tuple):
            tup = obj[0]
            name = str(tup[0]).strip()
            dose = float(tup[1]) if len(tup) > 1 else np.nan
            unit = str(tup[2]) if len(tup) > 2 else None
            return (name, dose, unit)
    except Exception:
        pass
    return (s, np.nan, None)

def m_support_stat(feature_set: set[str], index: dict[str, set[str]], train_n: int) -> tuple[float, float, float, int]:
    if not feature_set or train_n <= 0:
        return (0.0, 0.0, 0.0, 0)
    counts = np.array([len(index.get(feature, set())) for feature in feature_set], dtype=float)
    raw = float(np.mean(np.log1p(counts)))
    norm = raw / float(np.log1p(train_n))
    frac = float(np.mean(counts > 0))
    return (raw, norm, frac, int(len(feature_set)))

def m_feature_index(feature_map: dict[str, set[str]], train_genes: set[str]) -> dict[str, set[str]]:
    idx: dict[str, set[str]] = defaultdict(set)
    for gene in train_genes:
        for feature in feature_map.get(gene, set()):
            idx[feature].add(gene)
    return idx

def m_assign_bins(values: pd.Series, annot_count: pd.Series, cutpoints: list[float]) -> pd.Series:
    labels = ['L1_low', 'L2_mid', 'L3_high', 'L4_very_high']

    def label(v: object) -> str:
        if pd.isna(v):
            return 'L0_no_annotation'
        fv = float(v)
        for cp, lab in zip(cutpoints, labels):
            if fv <= cp:
                return lab
        return 'L4_very_high'
    out = values.map(label).astype(object)
    out[pd.to_numeric(annot_count, errors='coerce').fillna(0) <= 0] = 'L0_no_annotation'
    return out

5. Curves and controls¶

Calculate per-level means from the individual condition rows. The plots keep each representation and discovery/confirmation subset separate. Controls and contrasts below are calculated from the same rows. Higher overlap@N indicates better prediction.

Curve points are group means. A table column labelled standard deviation describes spread between records, not a confidence interval. Explicit bootstrap intervals, where present, resample condition rows and do not account for repeated genes across contexts or representations.

The adjusted few-shot regression uses context and representation indicators, without redundant subset indicators. Canonical-gene clusters preserve dependence across repeated contexts and representations. The 95% interval uses finite-sample-corrected cluster covariance and t(G−1) inference; the number of genes, design rank, exclusions and reference categories are displayed.         trend_rows.append({'spectra_subset': subset, 'representation': rep, 'n_levels': int(len(valid)), 'n_rows': int(valid['n_rows'].sum()), 'similarity_monotone_increasing': bool(valid['similarity_mean'].is_monotonic_increasing), 'spearman_level_vs_overlap_at_N': float(valid['level_order'].corr(valid['overlap_at_N_mean'], method='spearman')), 'low_level': low['level'], 'high_level': high['level'], 'low_overlap_at_N_mean': float(low['overlap_at_N_mean']), 'high_overlap_at_N_mean': float(high['overlap_at_N_mean']), 'high_minus_low_overlap_at_N': float(high['overlap_at_N_mean'] - low['overlap_at_N_mean'])})

def m_clustered_support_fit(df: pd.DataFrame, y_col: str, include_de_count: bool) -> dict[str, object]:
    from scipy.stats import t as student_t
    required = [y_col, 'support_norm_log_mean', 'feature_count', 'gene_symbol_canonical', 'context', 'representation']
    if include_de_count:
        required.append('de_nsig_counts_real')
    eligible = df[pd.to_numeric(df['feature_count'], errors='coerce') > 0].copy()
    work = eligible.dropna(subset=required).copy()
    numeric = [y_col, 'support_norm_log_mean', 'feature_count']
    if include_de_count:
        numeric.append('de_nsig_counts_real')
    work[numeric] = work[numeric].apply(pd.to_numeric, errors='coerce')
    work = work[np.isfinite(work[numeric]).all(axis=1)].copy()
    if include_de_count and (work['de_nsig_counts_real'] < 0).any():
        raise ValueError('DE counts cannot be negative')
    y = work[y_col].to_numpy(float)
    columns, names = [], []
    def add_standardized(name, values):
        values = np.asarray(values, dtype=float)
        sd = values.std(ddof=0)
        if not np.isfinite(sd) or sd <= 0:
            raise ValueError('Constant or invalid covariate: ' + name)
        columns.append((values - values.mean()) / sd)
        names.append(name)
    add_standardized('support_norm_log_mean_z', work['support_norm_log_mean'])
    add_standardized('feature_count_z', work['feature_count'])
    if include_de_count:
        add_standardized('log1p_de_nsig_counts_real_z', np.log1p(work['de_nsig_counts_real']))
    reference_categories = {}
    for col in ['context', 'representation']:
        categories = sorted(work[col].astype(str).unique())
        reference_categories[col] = categories[0]
        for value in categories[1:]:
            columns.append(work[col].astype(str).eq(value).to_numpy(float))
            names.append(col + '_' + value)
    x = np.column_stack([np.ones(len(work)), *columns])
    n, p = x.shape
    rank = int(np.linalg.matrix_rank(x))
    if rank != p or n <= p:
        raise ValueError('Regression design must have full column rank and residual degrees of freedom')
    groups, genes = pd.factorize(work['gene_symbol_canonical'].astype(str), sort=True)
    g = len(genes)
    if g < 2:
        raise ValueError('Gene-clustered inference requires at least two genes')
    beta = np.linalg.lstsq(x, y, rcond=None)[0]
    residual = y - np.einsum('ij,j->i', x, beta, optimize=False)
    # All observations of a canonical gene contribute to one cluster score.
    cluster_scores = np.zeros((g, p))
    np.add.at(cluster_scores, groups, x * residual[:, None])
    bread = np.linalg.inv(np.einsum('ij,ik->jk', x, x, optimize=False))
    correction = (g / (g - 1)) * ((n - 1) / (n - p))
    covariance = correction * np.einsum('ij,jk,kl->il', bread, np.einsum('ij,ik->jk', cluster_scores, cluster_scores, optimize=False), bread, optimize=False)
    se = np.sqrt(np.maximum(np.diag(covariance), 0))
    critical = float(student_t.ppf(.975, g - 1))
    ss_total = float(np.sum((y - y.mean()) ** 2))
    coefs = []
    for name, estimate, error in zip(['intercept', *names], beta, se):
        statistic = float(estimate / error) if error > 0 else np.nan
        coefs.append({'term': name, 'coef': float(estimate), 'se': float(error),
                      'ci_low': float(estimate - critical * error), 'ci_high': float(estimate + critical * error),
                      't': statistic, 'p_two_sided': float(2 * student_t.sf(abs(statistic), g - 1)),
                      'n_gene_clusters': g, 'cluster_df': g - 1, 'design_rank': rank,
                      'n_design_columns': p, 'n_excluded_missing': int(len(eligible) - n),
                      'covariance': 'CR1 canonical-gene clusters; t(G-1)',
                      'reference_categories': json.dumps(reference_categories, sort_keys=True)})
    return {'n_rows': n, 'n_gene_clusters': g, 'design_rank': rank,
            'r2': 1 - float(np.sum(residual ** 2)) / ss_total if ss_total else np.nan,
            'include_de_count': include_de_count, 'coefficients': coefs}
