def sha256_file(path):
    digest=hashlib.sha256()
    with Path(path).open('rb') as stream:
        for block in iter(lambda:stream.read(8*1024*1024),b''):digest.update(block)
    return digest.hexdigest()

def fetch_checked(url,destination,expected):
    destination=Path(destination)
    destination.parent.mkdir(parents=True,exist_ok=True)
    if not destination.is_file():
        temporary=destination.with_suffix(destination.suffix+'.part')
        with urllib.request.urlopen(url,timeout=120) as response,temporary.open('wb') as output:
            while True:
                block=response.read(1024*1024)
                if not block:break
                output.write(block)
        if sha256_file(temporary)!=expected:raise ValueError('Downloaded bytes failed SHA-256 verification: '+url)
        temporary.replace(destination)
    if sha256_file(destination)!=expected:raise ValueError('Cached bytes failed SHA-256 verification: '+str(destination))
    return destination

2. Data processing¶

Parse experimental chains, count resolved residues and compute sequence quantities: cysteine density, potential disulfide pairs, N-X-S/T glycosylation motifs, maximum19-residue Kyte–Doolittle average and maximum21-residue hydrophobic fraction.

from Bio.PDB import PDBParser,MMCIFParser
from Bio.SeqUtils import seq1

def read_reference_chain(path,chain_id):
    parser=MMCIFParser(QUIET=True) if Path(path).suffix=='.cif' else PDBParser(QUIET=True)
    structure=parser.get_structure('reference',str(path))
    model=next(structure.get_models())
    if str(chain_id) not in model:raise ValueError('Specified chain is absent: '+str(chain_id))
    residues=[residue for residue in model[str(chain_id)] if 'CA' in residue and residue.id[0] in (' ','H_MSE')]
    coordinates=np.array([residue['CA'].coord for residue in residues],dtype=float)
    sequence=''.join(seq1(residue.resname,custom_map={'MSE':'M'}) for residue in residues)
    if len(residues)==0:raise ValueError('No experimental C-alpha residues')
    return sequence,coordinates

def nglyc_count(seq: str) -> int:
    return len(re.findall('(?=N[^P][ST])', seq))

def disulfide_capacity_per100(seq: str) -> float:
    return seq.count('C') // 2 / (len(seq) / 100.0)

def tm_proxy(seq: str) -> float:
    kd = {'A': 1.8, 'C': 2.5, 'D': -3.5, 'E': -3.5, 'F': 2.8, 'G': -0.4, 'H': -3.2, 'I': 4.5, 'K': -3.9, 'L': 3.8, 'M': 1.9, 'N': -3.5, 'P': -1.6, 'Q': -3.5, 'R': -4.5, 'S': -0.8, 'T': -0.7, 'V': 4.2, 'W': -0.9, 'Y': -1.3}
    vals = [kd.get(aa, 0.0) for aa in seq]
    if len(vals) < 19:
        return sum(vals) / max(1, len(vals))
    return max((sum(vals[i:i + 19]) / 19.0 for i in range(len(vals) - 18)))

def max_hydrophobic_window(seq: str, width: int=21) -> float:
    hyd = set('AILMFWYV')
    if len(seq) <= width:
        return sum((aa in hyd for aa in seq)) / max(1, len(seq))
    return max((sum((aa in hyd for aa in seq[i:i + width])) / width for i in range(len(seq) - width + 1)))

3. Axis definition and level construction¶

Envelope/fusion level is the sum of disulfide density≥1.2 pairs/100 residues and the joint condition of hydropathy proxy≥1.8 with≤2 glycosylation motifs. The surface regime uses text-defined surface proteins excluding capsid/enzyme controls, disulfide density≥1.2 and hydropathy proxy<2.55. The code explicitly constructs all16 envelope candidate indicators and surface subtypes before joining outcomes.

def axis_features(row: dict) -> dict:
    text = ' '.join((str(row.get(c, '')) for c in ['organism', 'entity_description', 'title', 'keywords'])).lower()
    surface = contains(text, '\\b(envelope|spike|glycoprotein|fusion|gp1|gp2|gp350|protein e|e1|e2)\\b')
    capsid_or_enzyme = contains(text, '\\b(capsid|protease|helicase|polymerase|nsp5|ns2b|ns3)\\b')
    disulfide_dense = float(row['disulfide_capacity_per100']) >= 1.2
    low_helical = float(row['tm_helix_proxy_calc']) < 2.55
    if not surface or capsid_or_enzyme:
        level = 0
    elif disulfide_dense and low_helical:
        level = 2
    else:
        level = 1
    return {'surface_keyword': int(surface), 'capsid_or_enzyme_keyword': int(capsid_or_enzyme), 'surface_lowhelical_disulfide_dense': int(disulfide_dense), 'surface_lowhelical_low_tm_proxy': int(low_helical), 'surface_lowhelical_score': level, 'surface_lowhelical_level_name': {0: 'L0_non_surface_or_capsid_enzyme_control', 1: 'L1_surface_without_disulfide_lowhelical_combo', 2: 'L2_surface_disulfide_dense_lowhelical'}[level]}

def screen_norm_text(*values: object) -> str:
    return ' '.join(('' if pd.isna(v) else str(v).lower() for v in values))

def screen_contains(text: str, pattern: str) -> bool:
    return re.search(pattern, text, flags=re.I) is not None

def screen_add_features(df: pd.DataFrame) -> pd.DataFrame:
    out = df.copy()
    out['text'] = [screen_norm_text(a, b, c, d) for a, b, c, d in zip(out['organism'], out['entity_description'], out['title'], out['keywords'])]
    text = out['text']
    out['surface_keyword'] = text.map(lambda s: screen_contains(s, '\\b(envelope|spike|glycoprotein|fusion|gp1|gp2|gp350|protein e)\\b'))
    out['capsid_or_enzyme_keyword'] = text.map(lambda s: screen_contains(s, '\\b(capsid|protease|helicase|polymerase|nsp5|ns2b|ns3)\\b'))
    out['virion_vlp_context'] = text.map(lambda s: screen_contains(s, '\\b(virion|viron|vlp|virus-like|virus like|particle|inactivated|vaccine|asymmetric unit|prme|spike)\\b'))
    out['alphavirus_keyword'] = text.map(lambda s: screen_contains(s, '\\b(semliki|equine encephalitis|alphavirus)\\b'))
    out['flavivirus_keyword'] = text.map(lambda s: screen_contains(s, '\\b(tick-borne|yellow fever|japanese encephalitis|dengue|zika|langat|deer tick|orthoflavivirus|flavivirus|powassan)\\b'))
    out['class_ii_virus_keyword'] = out['alphavirus_keyword'] | out['flavivirus_keyword']
    out['class_ii_surface_name'] = text.map(lambda s: screen_contains(s, '\\b(envelope protein e|envelope protein|e glycoprotein|spike glycoprotein e1|spike glycoprotein e2|envelope glycoprotein e2|structural polyprotein|genome polyprotein)\\b'))
    out['e2_or_flavi_e_name'] = text.map(lambda s: screen_contains(s, '\\b(e2|envelope protein e|envelope protein|e glycoprotein)\\b'))
    out['fusion_f_or_s_name'] = text.map(lambda s: screen_contains(s, "\\b(fusion glycoprotein f1|fusion glycoprotein f2|spike protein s1|spike protein s2|spike protein s2')\\b"))
    out['gp1_gp2_g_name'] = text.map(lambda s: screen_contains(s, '\\b(gp1|gp2|glycoprotein g|gp350)\\b'))
    out['disulfide_dense'] = pd.to_numeric(out['disulfide_capacity_per100'], errors='coerce').ge(1.2)
    out['low_helical_surface_proxy'] = pd.to_numeric(out['tm_helix_proxy_calc'], errors='coerce').lt(2.55)
    out['very_high_hydrophobic_window'] = pd.to_numeric(out['hydrophobic_window21_max_frac_calc'], errors='coerce').ge(0.66)
    out['few_nglyc'] = pd.to_numeric(out['nglyc_motif_count'], errors='coerce').le(3)
    out['long_chain'] = pd.to_numeric(out['seq_len'], errors='coerce').ge(350)
    out['classII_beta_surface_seq'] = out['surface_keyword'] & out['class_ii_virus_keyword'] & out['disulfide_dense'] & out['low_helical_surface_proxy'] & out['few_nglyc'] & ~out['capsid_or_enzyme_keyword']

    diff_pred = pred[:, None, :] - pred[None, :, :]

    d_pred = model_scoring.np.linalg.norm(diff_pred, axis=-1)

def envelope_bootstrap_top_minus_lower(df: pd.DataFrame, rng_seed: int=0, n_boot: int=5000) -> dict:

    beta = np.linalg.lstsq(xv, yv, rcond=None)[0]

    dof = len(yv) - xv.shape[1]
    try:
        cov = resid @ resid / dof * np.linalg.inv(xv.T @ xv)
        se = float(np.sqrt(np.diag(cov))[1])
        t = float(beta[1] / se) if se else float('nan')
    except Exception:
        se = float('nan')
        t = float('nan')
    return {'n': int(len(yv)), 'coef': float(beta[1]), 'se': se, 't': t, 'controls': controls}

def screen_bootstrap_diff(df: pd.DataFrame, mask: pd.Series, metric: str, rng_seed: int=1, n_boot: int=5000) -> dict:
    rng = np.random.default_rng(rng_seed)
    top = pd.to_numeric(df.loc[mask, metric], errors='coerce').dropna().to_numpy(float)
    rest = pd.to_numeric(df.loc[~mask, metric], errors='coerce').dropna().to_numpy(float)
    if len(top) < 2 or len(rest) < 2:
        return {'n_top': int(len(top)), 'n_rest': int(len(rest)), 'mean': float('nan'), 'ci_low': float('nan'), 'ci_high': float('nan')}
    vals = [float(rng.choice(top, len(top), replace=True).mean() - rng.choice(rest, len(rest), replace=True).mean()) for _ in range(n_boot)]
    arr = np.asarray(vals)
    return {'n_top': int(len(top)), 'n_rest': int(len(rest)), 'mean': float(arr.mean()), 'ci_low': float(np.quantile(arr, 0.025)), 'ci_high': float(np.quantile(arr, 0.975))}
